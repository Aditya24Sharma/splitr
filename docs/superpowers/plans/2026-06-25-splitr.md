# Splitr Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A laptop-only Next.js web app that splits restaurant bills and per-item grocery receipts (parsed from photos by Gemini vision) into clean per-person amounts to copy into Splitwise.

**Architecture:** Single Next.js (App Router) app serving both the React/Tailwind UI and the API route handlers. All money is integer cents. The split math is a pure, dependency-free TypeScript module (`lib/`) that both the API and UI call, so previews equal stored results and the math is exhaustively unit-tested. Gemini parses receipt photos server-side into structured JSON; nothing persists until the operator saves. Prisma + SQLite store people and expense history in one local file.

**Tech Stack:** Next.js (App Router), React, TypeScript, Tailwind CSS, Prisma + SQLite, Google Gemini (`@google/genai`), `sharp` + `heic-convert` for image normalization, Vitest for tests.

## Global Constraints

- **Money is integer cents everywhere** except at UI/JSON display edges. No floats in math.
- **Single operator, no auth.** People are records, not users.
- **Laptop-only.** No Docker/hosting/cloud config. Runs via `npm run dev` at `http://localhost:3000`.
- **USD only.** No multi-currency.
- **Splits supported:** itemized (per-item even among assignees + prorated tax), simple even, simple exact. Nothing else.
- **Penny reconciliation = largest-remainder (Hamilton).** Rounded parts must sum exactly to the divided amount; deterministic tie-break by index.
- **`GEMINI_API_KEY`** read from `.env`; app fails with a clear message if missing.
- **Frontend design bans (hard):** no colored left-edge accent bars on cards; no uppercase wide letter-spacing labels; no colored-dot status pills; no ambient radial-gradient glows. Use plain 1px borders / background contrast, normal-case labels differentiated by weight/size, explicit text/icon status, flat surfaces.

```ts
// lib/money.ts
export type Cents = number

export function toCents(dollars: number): Cents {
  return Math.round((dollars + Number.EPSILON) * 100)
}
export function toDollars(cents: Cents): number {
  return cents / 100
}
export function formatUSD(cents: Cents): string {
  const neg = cents < 0
  const v = Math.abs(cents)
  return `${neg ? '-' : ''}$${Math.floor(v / 100)}.${String(v % 100).padStart(2, '0')}`
}
export function apportion(total: Cents, weights: number[]): Cents[] {
  if (weights.length === 0) throw new Error('apportion: weights must be non-empty')
  if (total < 0) throw new Error('apportion: total must be non-negative')
  const sum = weights.reduce((a, b) => a + b, 0)
  const useEqual = sum === 0
  const effWeights = useEqual ? weights.map(() => 1) : weights
  const effSum = useEqual ? weights.length : sum
  const raw = effWeights.map(w => (total * w) / effSum)
  const floors = raw.map(Math.floor)
  let remainder = total - floors.reduce((a, b) => a + b, 0)
  // distribute leftover cents to largest fractional remainders, tie-break by lower index
  const order = raw
    .map((r, i) => ({ i, frac: r - Math.floor(r) }))
    .sort((a, b) => (b.frac - a.frac) || (a.i - b.i))
  const result = floors.slice()
  for (let k = 0; k < remainder; k++) result[order[k].i] += 1
  return result
}
```

```ts
// lib/split.ts
import { apportion, Cents } from '@/lib/money'

export interface PersonShare { personId: string; cents: Cents }
export interface SplitItem { id: string; lineTotalCents: Cents; assigneeIds: string[] }
export interface ItemizedInput { items: SplitItem[]; taxCents: Cents }
export interface ItemizedResult {
  perPerson: PersonShare[]
  perItem: { itemId: string; effectiveCents: Cents }[]
  totalCents: Cents
}

function gcd(a: number, b: number): number { while (b) { [a, b] = [b, a % b] } return a }
function lcm(a: number, b: number): number { return (a / gcd(a, b)) * b }

export function splitItemized(input: ItemizedInput): ItemizedResult {
  const { items, taxCents } = input
  if (items.length === 0) throw new Error('splitItemized: no items')
  for (const it of items) {
    if (it.assigneeIds.length === 0) throw new Error(`splitItemized: item ${it.id} has no assignees`)
  }
  // 1. Prorate tax across items by line-total share; each item's effective (tax-inclusive) cost.
  const taxParts = apportion(taxCents, items.map(i => i.lineTotalCents))
  const perItem = items.map((it, idx) => ({ itemId: it.id, effectiveCents: it.lineTotalCents + taxParts[idx] }))
  const totalCents = perItem.reduce((a, b) => a + b.effectiveCents, 0)
  // 2. Each person's EXACT fractional share across the whole receipt, kept integer by
  //    scaling with D = lcm of assignee counts (scaled value is the numerator over D).
  const D = items.reduce((acc, it) => lcm(acc, it.assigneeIds.length), 1)
  const scaled = new Map<string, number>()
  items.forEach((it, idx) => {
    const perAssignee = perItem[idx].effectiveCents * (D / it.assigneeIds.length) // integer
    for (const pid of it.assigneeIds) scaled.set(pid, (scaled.get(pid) ?? 0) + perAssignee)
  })
  // 3. One largest-remainder reconciliation at the PERSON level => fairest split that
  //    sums exactly to totalCents; deterministic tie-break by sorted personId.
  const persons = [...scaled.keys()].sort()
  const floors = persons.map(p => Math.floor(scaled.get(p)! / D))
  const leftover = totalCents - floors.reduce((a, b) => a + b, 0)
  const order = persons
    .map((p, i) => ({ i, rem: scaled.get(p)! % D }))
    .sort((a, b) => (b.rem - a.rem) || (a.i - b.i))
  const cents = floors.slice()
  for (let k = 0; k < leftover; k++) cents[order[k].i] += 1
  const perPerson = persons.map((personId, i) => ({ personId, cents: cents[i] }))
  return { perPerson, perItem, totalCents }
}

export function splitSimpleEven(totalCents: Cents, participantIds: string[]): PersonShare[] {
  if (participantIds.length === 0) throw new Error('splitSimpleEven: no participants')
  const parts = apportion(totalCents, participantIds.map(() => 1))
  return participantIds.map((personId, i) => ({ personId, cents: parts[i] }))
}

export function splitSimpleExact(totalCents: Cents, shares: PersonShare[]): PersonShare[] {
  const s = shares.reduce((a, b) => a + b.cents, 0)
  if (s !== totalCents) throw new Error('exact split must sum to total')
  return shares
}
```

```ts
export async function normalizeImage(buffer: Buffer, mime: string): Promise<Buffer> {
  let input = buffer
```

```ts
import { GoogleGenAI, Type } from '@google/genai'
import { ParsedReceipt } from '@/lib/types'

const MODEL = process.env.GEMINI_MODEL || 'gemini-2.5-flash'
const SYSTEM = `You extract structured data from US grocery receipts (Walmart, Sam's Club).
Expand cryptic item abbreviations into readable names (e.g. "GVGRK LT VA" -> "Great Value Greek Light Vanilla Yogurt").
Resolve weighted lines ("1.67 lb @ 1.27/lb") and multi-buy ("2 AT 1 FOR 0.97") into quantity, unitPrice, lineTotal.
Combine nothing; one row per printed line item. All prices in dollars. Use null when unknown.`

const SCHEMA = {
  type: Type.OBJECT,
  properties: {
    store: { type: Type.STRING, nullable: true },
    purchasedAt: { type: Type.STRING, nullable: true },
    items: { type: Type.ARRAY, items: { type: Type.OBJECT, properties: {
      rawName: { type: Type.STRING, nullable: true },
      name: { type: Type.STRING },
      quantity: { type: Type.NUMBER },
      unitPrice: { type: Type.NUMBER },
      lineTotal: { type: Type.NUMBER },
    }, required: ['name', 'quantity', 'unitPrice', 'lineTotal'] } },
    subtotal: { type: Type.NUMBER }, tax: { type: Type.NUMBER }, total: { type: Type.NUMBER },
    warnings: { type: Type.ARRAY, items: { type: Type.STRING } },
  },
  required: ['items', 'subtotal', 'tax', 'total'],
}

export function validateParsedReceipt(r: ParsedReceipt): string[] {
  const w: string[] = []
  const c = (n: number) => Math.round(n * 100)
  const itemsSum = r.items.reduce((a, i) => a + c(i.lineTotal), 0)
  if (Math.abs(itemsSum - c(r.subtotal)) > 3) w.push(`Items ($${(itemsSum/100).toFixed(2)}) don't match subtotal ($${r.subtotal.toFixed(2)})`)
  if (Math.abs(c(r.subtotal) + c(r.tax) - c(r.total)) > 3) w.push(`Subtotal + tax doesn't match total ($${r.total.toFixed(2)})`)
  return w
}

export async function parseReceiptImage(jpeg: Buffer): Promise<ParsedReceipt> {
  const apiKey = process.env.GEMINI_API_KEY
  if (!apiKey) throw new Error('GEMINI_API_KEY missing')
  const ai = new GoogleGenAI({ apiKey })
  const res = await ai.models.generateContent({
    model: MODEL,
    contents: [{ role: 'user', parts: [
      { inlineData: { mimeType: 'image/jpeg', data: jpeg.toString('base64') } },
      { text: 'Extract this receipt as JSON.' },
    ] }],
    config: { systemInstruction: SYSTEM, responseMimeType: 'application/json', responseSchema: SCHEMA },
  })
  const parsed = JSON.parse(res.text) as ParsedReceipt
  parsed.warnings = [...(parsed.warnings ?? []), ...validateParsedReceipt(parsed)]
  return parsed
}
```

import { NextRequest, NextResponse } from 'next/server'
import { normalizeImage } from '@/lib/image'
import { parseReceiptImage } from '@/lib/gemini'

export async function POST(req: NextRequest) {
  try {
    const form = await req.formData()
    const file = form.get('image')
    if (!(file instanceof File)) return NextResponse.json({ error: 'No image uploaded' }, { status: 400 })
    const buf = Buffer.from(await file.arrayBuffer())
    const jpeg = await normalizeImage(buf, file.type)
    const parsed = await parseReceiptImage(jpeg)
    return NextResponse.json(parsed)
  } catch (e) {
    return NextResponse.json({ error: e instanceof Error ? e.message : 'Parse failed' }, { status: 500 })
  }
}
